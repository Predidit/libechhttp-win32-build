cmake_minimum_required(VERSION 3.22)
project(echhttp_dependencies LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
if(NOT DEFINED EH_CURL_SOURCE OR NOT DEFINED EH_BORINGSSL_SOURCE OR NOT DEFINED EH_BORINGSSL_BUILD)
  message(FATAL_ERROR "Use the Dart build hook to supply pinned dependencies")
endif()

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_CURL_EXE OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_LIBCURL_DOCS OFF CACHE BOOL "" FORCE)
set(BUILD_MISC_DOCS OFF CACHE BOOL "" FORCE)
set(ENABLE_CURL_MANUAL OFF CACHE BOOL "" FORCE)
set(CURL_USE_OPENSSL ON CACHE BOOL "" FORCE)
set(CURL_USE_SCHANNEL OFF CACHE BOOL "" FORCE)
set(CURL_USE_LIBPSL OFF CACHE BOOL "" FORCE)
set(CURL_USE_LIBSSH2 OFF CACHE BOOL "" FORCE)
set(CURL_USE_LIBSSH OFF CACHE BOOL "" FORCE)
set(CURL_USE_GSSAPI OFF CACHE BOOL "" FORCE)
set(CURL_ZLIB ON CACHE BOOL "" FORCE)
set(CURL_BROTLI OFF CACHE BOOL "" FORCE)
set(CURL_ZSTD OFF CACHE BOOL "" FORCE)
set(USE_NGHTTP2 OFF CACHE BOOL "" FORCE)
set(USE_LIBIDN2 OFF CACHE BOOL "" FORCE)
set(USE_ECH ON CACHE BOOL "" FORCE)
set(CURL_CA_BUNDLE "none" CACHE STRING "" FORCE)
set(CURL_CA_PATH "none" CACHE STRING "" FORCE)
set(CURL_CA_FALLBACK OFF CACHE BOOL "" FORCE)
set(HTTP_ONLY ON CACHE BOOL "" FORCE)
set(OPENSSL_INCLUDE_DIR "${EH_BORINGSSL_SOURCE}/include" CACHE PATH "" FORCE)
if(WIN32)
  set(OPENSSL_SSL_LIBRARY "${EH_BORINGSSL_BUILD}/ssl.lib" CACHE FILEPATH "" FORCE)
  set(OPENSSL_CRYPTO_LIBRARY "${EH_BORINGSSL_BUILD}/crypto.lib" CACHE FILEPATH "" FORCE)
  # FindOpenSSL selects these MSVC-specific variables before setting the
  # generic OPENSSL_* variables, even for a single-config Ninja build.
  set(LIB_EAY_RELEASE "${OPENSSL_CRYPTO_LIBRARY}" CACHE FILEPATH "" FORCE)
  set(LIB_EAY_DEBUG "${OPENSSL_CRYPTO_LIBRARY}" CACHE FILEPATH "" FORCE)
  set(SSL_EAY_RELEASE "${OPENSSL_SSL_LIBRARY}" CACHE FILEPATH "" FORCE)
  set(SSL_EAY_DEBUG "${OPENSSL_SSL_LIBRARY}" CACHE FILEPATH "" FORCE)
else()
  set(OPENSSL_SSL_LIBRARY "${EH_BORINGSSL_BUILD}/libssl.a" CACHE FILEPATH "" FORCE)
  set(OPENSSL_CRYPTO_LIBRARY "${EH_BORINGSSL_BUILD}/libcrypto.a" CACHE FILEPATH "" FORCE)
endif()
# BoringSSL is C++ internally. curl's configure checks link C executables;
# explicitly propagate the target toolchain's C++ runtime to those checks.
# This selects libc++ on Apple/Android and libstdc++ on GCC Linux.
find_package(OpenSSL REQUIRED)
if(ANDROID)
  # -lc++ resolves to the NDK's shared linker script even with
  # ANDROID_STL=c++_static. Keep the C probe and final asset self-contained.
  set_property(TARGET OpenSSL::SSL APPEND PROPERTY INTERFACE_LINK_LIBRARIES c++_static c++abi ${CMAKE_DL_LIBS})
elseif(NOT MSVC)
  set_property(TARGET OpenSSL::SSL APPEND PROPERTY INTERFACE_LINK_LIBRARIES ${CMAKE_CXX_IMPLICIT_LINK_LIBRARIES})
endif()
if(NOT HAVE_SSL_SET1_ECH_CONFIG_LIST)
  unset(HAVE_SSL_SET1_ECH_CONFIG_LIST CACHE)
endif()
if(NOT HAVE_SSL_SET0_WBIO)
  unset(HAVE_SSL_SET0_WBIO CACHE)
endif()
add_subdirectory("${EH_CURL_SOURCE}" curl EXCLUDE_FROM_ALL)

add_library(echhttp_deps_probe SHARED probe.cpp)
target_link_libraries(echhttp_deps_probe PRIVATE CURL::libcurl)
target_include_directories(echhttp_deps_probe PRIVATE "${EH_BORINGSSL_SOURCE}/include")
add_executable(echhttp_deps_check EXCLUDE_FROM_ALL probe.cpp)
target_compile_definitions(echhttp_deps_check PRIVATE EH_PROBE_EXECUTABLE)
target_link_libraries(echhttp_deps_check PRIVATE CURL::libcurl)
target_include_directories(echhttp_deps_check PRIVATE "${EH_BORINGSSL_SOURCE}/include")
if(ANDROID)
  target_link_options(echhttp_deps_probe PRIVATE "-Wl,-z,max-page-size=16384")
endif()
